<script>
  function showCat(catId, btn) {
    document.querySelectorAll('.cat-section').forEach(s => s.classList.add('hidden'));
    document.getElementById('cat-' + catId).classList.remove('hidden');
    document.querySelectorAll('.cat-btn').forEach(b => b.classList.remove('active'));
    btn.classList.add('active');
  }

  function pedir(item) {
    const numero = '5511957283370';
    let msg;
    if (item) {
      msg = 'Oi! Vi no cardápio digital e quero pedir: ' + item;
    } else {
      msg = 'Oi! Vi o cardápio digital e quero fazer um pedido no HICK.LANSHI 🍔';
    }
    const url = 'https://wa.me/' + numero + '?text=' + encodeURIComponent(msg);
    window.open(url, '_blank');
  }
